## 二、数据库 Schema

### 2.1 用户表 (users)

```sql
CREATE TABLE users (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nickname      VARCHAR(100),
  phone         VARCHAR(20),       -- 留资用
  email         VARCHAR(255),
  source        VARCHAR(50),        -- 渠道来源: webchat/wechat/qrcode/community
  created_at    TIMESTAMP DEFAULT NOW(),
  updated_at    TIMESTAMP DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_users_phone ON users(phone);
CREATE INDEX idx_users_email ON users(email);
```

### 2.2 诊断记录表 (diagnosis_records)

```sql
CREATE TABLE diagnosis_records (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id       UUID NOT NULL REFERENCES users(id),
  type          VARCHAR(10) NOT NULL CHECK (type IN ('child', 'adult', 'senior')),
  version       VARCHAR(20) DEFAULT '1.0',   -- 题库版本号，便于版本回溯
  status        VARCHAR(20) DEFAULT 'completed' CHECK (status IN ('in_progress', 'completed', 'abandoned')),
  
  -- 诊断结果（来自Scorer的输出）
  result_json   JSONB NOT NULL,               -- DiagnosisResult 完整对象
  
  -- 对标元数据（Phase1预铺，对海外对标数据）
  benchmark_meta JSONB DEFAULT NULL,           -- 见下文详细定义
  
  -- 用户基础信息（非必填，留资用）
  user_name     VARCHAR(50),
  user_age      INT,
  user_gender   VARCHAR(10),
  user_occupation VARCHAR(100),               -- 成人版：职业
  user_industry   VARCHAR(100),               -- 成人版：行业
  
  -- 评分器元数据
  scorer_version VARCHAR(50),                 -- 评分器算法版本号
  question_ids   JSONB,                       -- 本次使用的题目ID列表
  
  -- 时间字段
  started_at    TIMESTAMP DEFAULT NOW(),
  completed_at  TIMESTAMP,
  created_at    TIMESTAMP DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_diag_user_id ON diagnosis_records(user_id);
CREATE INDEX idx_diag_type ON diagnosis_records(type);
```


### 3.3 Scorer 接口（评分器插拔）

```typescript
// /engine/scorer.ts

export interface Scorer {
  readonly type: 'child' | 'adult' | 'senior';
  readonly version: string;       // 语义化版本号 "1.0.0"
  
  // 核心评分方法
  score(input: ScoreInput): ScoreOutput;
  
  // 附加能力
  getDimensionDefinitions(): DimensionDefinition[];
  getQuestionSelector(): QuestionSelector;
}

export interface ScoreInput {
  questions: Question[];         // 用户看到的题目
  answers: AnswerRecord[];       // 用户的选择
  userInfo?: {
    age?: number;
    occupation?: string;
    industry?: string;
  };
}

export interface ScoreOutput {
  dimensions: DimensionScore[];
  
  // 儿童/成人：总分（0-100）
  overallScore?: number;
  
  // 老年人：场景匹配度列表（推荐/可选/不推荐）
  scenarioRecommendations?: ScenarioRecommendation[];
  
  // 安全维度聚合（出海用，Phase2+启用）
  securityScore?: {
    overall: number;
    breakdown: Record<string, number>;
  };
  
  // 元数据
  scorerVersion: string;
  scoringTimestamp: string;
  benchmarkMeta?: BenchmarkMeta;  // 对标元数据
}

export interface DimensionScore {
  dimensionId: string;            // 简称ID
  name: string;                   // 中文名（前端展示用）
  nameEn: string;                 // 英文名（前端展示用）
  score: number | null;           // 儿童/成人：0-100；老年人：null
  matchLevel?: number;            // 老年人：0-1
  label?: 'excellent' | 'good' | 'needs_improvement' | 'recommended' | 'optional';
  scoreHistory?: number[];        // 历史得分（复测用）
  suggestions: string[];          // 建议列表
  securityTag?: boolean;          // 是否为安全相关子指标
}
```

## 五、Scorer 评分器：3个实现

### 5.1 ChildScorer — 5维加权算法

```typescript
class ChildScorer implements Scorer {
  type = 'child' as const;
  version = '1.0.0';
  
  score(input: ScoreInput): ScoreOutput {
    // 对每个维度：
    // 1. 收集该维度下的所有题目得分
    // 2. 加权平均（难度越高权重越大）
    // 3. 归一化到0-100
    //
    // weight = easy × 1.0 + medium × 1.5 + hard × 2.0
    // dimensionScore = Σ(题得分 × weight) / Σ(maxScore × weight) × 100
    
    // 5维输出 + 雷达图label
  }
}
```

### 5.2 AdultScorer — 8维三层算法

```typescript
class AdultScorer implements Scorer {
  type = 'adult' as const;
  version = '1.0.0';
  
  score(input: ScoreInput): ScoreOutput {
    // 对每个维度：
    // 1. 同ChildScorer加权算法
    // 2. 按三层分组展示
    //    - baseLayer: D01+D02+D03
    //    - advancedLayer: D04+D05+D06
    //    - leadershipLayer: D07+D08
    
    // 特殊处理：D08 AI元认知使用不同视觉标记（虚线/内圈）
    // SecurityScore聚合（Phase2启用）
  }
}
```

### 5.3 SeniorMatcher — 场景匹配逻辑

```typescript
class SeniorMatcher implements Scorer {
  type = 'senior' as const;
  version = '1.0.0';
  
  score(input: ScoreInput): ScoreOutput {
    // 需求选择模式：用户从4场景中选择1个
    // 不计算分数
    // 输出：选中的场景 → 推荐场景 → 视频URL → 下次推送间隔
    
    // 行为追踪：
    // 1. 记录用户选择了哪个场景
    // 2. 记录用户是否完成视频
    // 3. 记录用户是否分享到家庭群
    // 4. 计算用户对该场景的"完成度"（0-1）
    // 5. 根据完成度决定下次推送什么场景
    
    // 如果完成度 > 0.8 → 推送下一个场景
    // 如果完成度 < 0.5 → 重复推送同一场景（减少新学习负担）
    // 如果3天无操作 → 推送微信模板消息给子女
  }
}
```

## 七、题库路由策略

### 7.1 自动出题逻辑

```typescript
class QuestionSelector {
  // 用户诊断时的选题策略
  
  selectQuestions(
    bank: Question[],
    type: 'child' | 'adult',
    userInfo?: { age?: number; occupation?: string; }
  ): Question[] {
    // Step 1: 按维度分组（8维/5维）
    // Step 2: 每个维度抽 2-3 题
    //   - 优先抽 medium 难度（最佳区分度）
    //   - 如果 medium 不足，补 easy（低分用户区分）
    //   - 每维度最多1题hard（高分用户区分）
    // Step 3: 总题数控制
    //   - MVP版：8维×2-3题 = 16-24题（推荐20题）
    //   - 完整版：8维×5题 = 40题
    // Step 4: 去重 + 乱序
    // Step 5: 返回题目列表
  }
}
```

### 7.2 路由分发入口

```typescript
// /api/v1/diagnosis/start
async function handleDiagnosisStart(req: StartDiagnosisRequest): StartDiagnosisResponse {
  const bank = getQuestionBank(req.type);     // 选择题库池
  const questions = questionSelector.selectQuestions(bank.questions, req.type);
  
  const diagnosis = await createDiagnosisRecord({
    userId: req.userId,
    type: req.type,
    questionIds: questions.map(q => q.id),
  });
  
  return {
    diagnosisId: diagnosis.id,
    type: req.type,
    questions,
    totalQuestions: questions.length,
    estimatedMinutes: questions.length * 0.9,  // 约54秒/题
  };
}

function getQuestionBank(type: string): QuestionBank {
  switch (type) {
    case 'child':  return childBank;    // CC-* 编码
    case 'adult':  return adultBank;    // AC-* 编码
    case 'senior': return seniorBank;   // SC-* 编码
    default: throw new Error(`Unknown diagnosis type: ${type}`);
  }
}
```
